# GUI / Wayland session packages. Installed only for profile=gui.

# --- compositor + helpers ---
sway                         # Wayland-композитор (i3-подобный; конфиг sway/)
niri                         # Wayland-композитор со скроллящимися колонками (конфиг niri/)
waybar                       # панель статуса для Wayland
xwayland-satellite           # Xwayland-сервер для niri
swaybg                       # установка обоев рабочего стола
swayidle                     # реакция на простой (блокировка/гашение экрана)
swaylock                     # экран блокировки
swaync                       # центр уведомлений
gammastep                    # сдвиг цветовой температуры (night light)
brightnessctl                # управление яркостью подсветки
geoclue                      # служба геолокации (для гаммы по солнцу)
mesa-utils                   # утилиты Mesa (glxinfo, eglinfo)
vulkan-mesa-layers           # слои Vulkan для драйверов Mesa
radeontop                    # мониторинг GPU AMD

# --- terminal / launcher ---
alacritty                    # терминал на GPU (конфиг alacritty/)
rofi                         # лаунчер и меню (конфиг rofi/)
rofi-emoji                   # плагин эмодзи для rofi

# --- theme / appearance ---
darkman                      # автопереключение светлой/тёмной темы
kvantum                      # темизация Qt-приложений
arc-gtk-theme                # GTK-тема Arc
arc-icon-theme               # набор иконок Arc
adw-gtk-theme                # тема GTK-приложений в стиле Adwaita

# --- fonts ---
noto-fonts                   # базовые шрифты Noto
noto-fonts-emoji             # шрифт эмодзи Noto Color Emoji
adobe-source-code-pro-fonts  # моноширинный шрифт Source Code Pro
ttf-dejavu-nerd              # Nerd Font на базе DejaVu
ttf-jetbrains-mono           # шрифт JetBrains Mono
ttf-roboto                   # шрифт Roboto
terminus-font                # моноширинный шрифт Terminus (консоль)

# --- network ---
networkmanager               # демон управления сетью
networkmanager-openvpn       # плагин OpenVPN для NetworkManager (см. vpn/)
network-manager-applet       # трей-апплет NetworkManager
libnma-gtk4                  # библиотека диалогов NetworkManager (GTK4)
blueman                      # GUI для управления Bluetooth
bluez-utils                  # утилиты Bluetooth (bluetoothctl)

# --- audio ---
pipewire                     # звуковой сервер
pipewire-alsa                # поддержка ALSA через PipeWire
pipewire-pulse               # совместимость с PulseAudio
wireplumber                  # менеджер сессий PipeWire
pavucontrol                  # микшер и управление звуком (GUI)

# --- wayland utils / portals / notifications ---
wl-clipboard                 # буфер обмена в Wayland (wl-copy/wl-paste)
wtype                        # эмуляция ввода с клавиатуры в Wayland
grim                         # скриншоты в Wayland
slurp                        # выбор области экрана (для grim)
libnotify                    # отправка уведомлений (notify-send)
xorg-xwayland                # сервер X11 для запуска X-приложений
xdg-desktop-portal           # шлюз к системным сервисам для приложений
xdg-desktop-portal-gtk       # GTK-бэкенд порталов
xdg-desktop-portal-gnome     # GNOME-бэкенд порталов
