# Official-repo packages installed on EVERY host (including the headless one).
# One package per line; '#' starts a comment.
#   gui.txt  -> Wayland/GUI only
#   dev.txt  -> dev/DevOps tooling (gui profile; opt-in for headless hosts)
#   hosts/<host>/packages.txt -> host-specific extras

# --- base system / build ---
base                        # минимальная базовая система Arch
base-devel                  # инструменты сборки пакетов (gcc, make, patch, fakeroot)
git                         # распределённая система контроля версий

git-delta                   # подсветка diff для git и lazygit
openssh                     # SSH-клиент и сервер, ssh-keygen
rsync                       # инкрементальная синхронизация файлов
curl                        # HTTP-клиент для терминала
wget                        # загрузчик файлов по сети
jq                          # обработка JSON из командной строки
tree                        # вывод дерева каталогов
less                        # постраничный просмотр файлов (pager)
man-db                      # система man-страниц
unzip                       # распаковка .zip
7zip                        # архиватор (7z, zip, rar и др.)
unrar                       # распаковка .rar
htop                        # интерактивный монитор процессов
pkgconf                     # утилита pkg-config для флагов сборки
make                        # сборка проектов по Makefile
cmake                       # генератор систем сборки (C/C++)
clang                       # компилятор C/C++ из LLVM
lld                         # быстрый линкер LLVM

# --- shell / prompt / navigation ---
fish                        # интерактивный shell (основной в этом репо)

fd                          # быстрый аналог find

direnv                      # автозагрузка переменных окружения из .envrc
yazi                        # TUI-файловый менеджер

vim-devicons                # иконки для файловых плагинов vim
marksman                    # LSP для Markdown
tombi                       # LSP и форматтер для TOML
gopls                       # официальный LSP для Go
rust-analyzer               # LSP для Rust
bash-language-server        # LSP для Bash
dockerfile-language-server  # LSP для Dockerfile
vscode-json-languageserver  # LSP для JSON
yaml-language-server        # LSP для YAML
prettier                    # форматтер JS/TS/JSON/CSS/Markdown
mermaid-cli                 # рендер диаграмм Mermaid (mmdc)

go                          # язык Go и его toolchain
rustup                      # менеджер toolchain Rust
cargo-expand                # показ макросов после раскрытия (cargo expand)
nodejs                      # среда выполнения JavaScript
npm                         # менеджер пакетов Node.js
